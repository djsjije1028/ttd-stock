    <canvas id="chart"></canvas>
  </div>

  <script>
    let chart;

    async function loadPrice() {
      const res = await fetch("/price");
      const data = await res.json();

      document.getElementById("price").innerText =
        data.price.toLocaleString();

      if (!chart) {
        const ctx = document.getElementById("chart").getContext("2d");
        chart = new Chart(ctx, {
          type: "line",
          data: {
            labels: data.history.map((_, i) => i + 1),
            datasets: [{
              label: "TTD주식",
              data: data.history,
              borderColor: "#ffd400",
              backgroundColor: "rgba(255,212,0,0.15)",
              tension: 0.3
            }]
          },
          options: {
            plugins: {
              legend: { labels: { color: "#ffd400" } }
            },
            scales: {
              x: { ticks: { color: "#ffd400" } },
              y: { ticks: { color: "#ffd400" } }
            }
          }
        });
      } else {
        chart.data.labels = data.history.map((_, i) => i + 1);
        chart.data.datasets[0].data = data.history;
        chart.update();
      }
    }

    loadPrice();
  </script>
